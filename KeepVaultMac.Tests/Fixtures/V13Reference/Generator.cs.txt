static byte[] B(int n,int m,int s)=>Enumerable.Range(0,n).Select(i=>unchecked((byte)(i*m+s))).ToArray();

static byte[] LP(params byte[][] parts){using var m=new MemoryStream();using var w=new BinaryWriter(m);foreach(var p in parts){w.Write(p.Length);w.Write(p);}return m.ToArray();}
static byte[] Sha(byte[] x){var h=new Sha3Digest(512);h.BlockUpdate(x);var o=new byte[64];h.DoFinal(o);return o;}

static byte[] Argon(byte[] password,byte[] salt,byte[] secret,byte[] ad,int mem){var b=new Argon2Parameters.Builder(Argon2Parameters.Argon2id).WithVersion(Argon2Parameters.Version13).WithIterations(4).WithParallelism(4).WithMemoryAsKB(mem).WithSalt(salt).WithAdditional(ad);if(secret.Length>0)b.WithSecret(secret);var g=new Argon2BytesGenerator();g.Init(b.Build());var o=new byte[64];g.GenerateBytes(password,o);return o;}
static byte[] RawSkein(byte[] k,byte[] x){var h=new SkeinMac(1024,1024);h.Init(new KeyParameter(k));h.BlockUpdate(x);var o=new byte[128];h.DoFinal(o);return o;}
static (byte[],byte[]) Tree(byte[][] prefixes,byte[] ciphertext,byte[] shaKey,byte[] skKey){

using var stream=new MemoryStream();stream.Write(U(dom+"Root"));var header=new byte[20];long leaves=(logical.LongLength+1048575)/1048576;BinaryPrimitives.WriteInt64BigEndian(header,logical.LongLength);BinaryPrimitives.WriteInt64BigEndian(header.AsSpan(8),leaves);BinaryPrimitives.WriteInt32BigEndian(header.AsSpan(16),1048576);stream.Write(header);
