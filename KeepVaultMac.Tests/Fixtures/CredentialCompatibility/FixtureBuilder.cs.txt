static string HashFile(string path) { using var input = File.OpenRead(path); return Convert.ToHexString(SHA256.HashData(input)); }
static byte[] Frame(params byte[][] fields) {
    using var stream = new MemoryStream();
    foreach (byte[] field in fields) { byte[] length = new byte[4]; BinaryPrimitives.WriteInt32LittleEndian(length, field.Length); stream.Write(length); stream.Write(field); }
    return stream.ToArray();
}
static byte[] Digest(byte[] message) { var d = new Sha3Digest(512); d.BlockUpdate(message); byte[] result = new byte[64]; d.DoFinal(result); return result; }
static byte[] Sha3Reference(string algorithm, string password, string pin, byte[] a, byte[] b) {
    byte[] p = Encoding.UTF8.GetBytes(password), n = Encoding.ASCII.GetBytes(pin);
    return [..Digest(Frame(Encoding.UTF8.GetBytes($"Kalyna-ZPAQ/v12/{algorithm}/SHA3-512/User+PIN+Factors-A1+B1"), p, n, a[..64], b[..64])),
            ..Digest(Frame(Encoding.UTF8.GetBytes($"Kalyna-ZPAQ/v12/{algorithm}/SHA3-512/User+PIN+Factors-A2+B2"), p, n, a[64..], b[64..]))];
}
static byte[] SkeinReference(string algorithm, string password, string pin, byte[] a, byte[] b) {
    var mac = new SkeinMac(1024, 1024);
    mac.Init(new SkeinParameters.Builder().SetKey([..a, ..b]).SetPersonalisation(Encoding.UTF8.GetBytes($"Kalyna-ZPAQ/v12/{algorithm}/Skein-MAC-1024-1024/User+PIN/Factors-A+B-Key")).Build());
    mac.BlockUpdate(Frame(Encoding.UTF8.GetBytes(password), Encoding.ASCII.GetBytes(pin)));
    byte[] result = new byte[128]; mac.DoFinal(result); return result;
}
static GeneratedArchiveEntropy Entropy(string a, string b, int index) {
    var salt1 = LockedSensitiveBuffer.Create(EntropyMixer.SaltPairBytes); var nonce1 = LockedSensitiveBuffer.Create(EncryptionSuiteCatalog.MaxNonceBytes);
    var salt2 = LockedSensitiveBuffer.Create(EntropyMixer.SaltPairBytes); var nonce2 = LockedSensitiveBuffer.Create(EncryptionSuiteCatalog.MaxNonceBytes);
    Fill(salt1.Bytes, 17 + index, 29); Fill(nonce1.Bytes, 43 + index, 43); Fill(salt2.Bytes, 161 + index, 61); Fill(nonce2.Bytes, 211 + index, 73);
    return new GeneratedArchiveEntropy(a, b, salt1, nonce1, salt2, nonce2);
}
static void Fill(byte[] bytes, int seed, int multiplier) { for (int i=0; i<bytes.Length; i++) bytes[i] = (byte)(seed+i*multiplier); }
